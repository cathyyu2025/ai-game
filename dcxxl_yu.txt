    <script>
        // ---------- 数据 ----------
        let wordPairs = [];
        let remainingPairs = [];    // 当前游戏中剩余未匹配的配对
        let selectedCard = null;
        let startTime, timerInterval, correctCount = 0;
        let gameRunning = false;

        const defaultWords = [
            {word: "ありがとう", meaning: "谢谢"},
            {word: "こんにちは", meaning: "你好"},
            {word: "さようなら", meaning: "再见"},
            {word: "おやすみ", meaning: "晚安"},
            {word: "すみません", meaning: "对不起"},
            {word: "おはよう", meaning: "早上好"},
            {word: "こんばんは", meaning: "晚上好"},
            {word: "いただきます", meaning: "我开动了"},
            {word: "ごちそうさま", meaning: "多谢款待"},
            {word: "お願いします", meaning: "拜托了"}
        ];

        // 初始化排行榜
        (function init() {
            if (!localStorage.getItem('wordGameRecords')) localStorage.setItem('wordGameRecords', JSON.stringify([]));
            showLeaderboard();
        })();

        // 文件导入
        document.getElementById('fileInput').addEventListener('change', handleFile);
        function handleFile(e) {
            const file = e.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const data = new Uint8Array(e.target.result);
                    const workbook = XLSX.read(data, { type: 'array', codepage: 65001 });
                    const sheetName = workbook.SheetNames.find(name => {
                        const sheet = workbook.Sheets[name];
                        return XLSX.utils.sheet_to_json(sheet).length > 0;
                    }) || workbook.SheetNames[0];
                    const sheet = workbook.Sheets[sheetName];
                    const jsonData = XLSX.utils.sheet_to_json(sheet, {
                        header: ['word', 'meaning'],
                        range: 0,
                        blankrows: false,
                        defval: ""
                    });
                    wordPairs = jsonData
                        .filter(row => row.word && row.meaning)
                        .map(row => ({
                            word: sanitizeText(row.word),
                            meaning: sanitizeText(row.meaning)
                        }));
                    if (wordPairs.length === 0) throw new Error('有効なデータが見つかりません');
                    showImportedWords();
                    alert(`成功导入 ${wordPairs.length} 组单词！`);
                } catch (error) {
                    alert(`ファイルの読み込みに失敗しました: ${error.message}`);
                }
            };
            reader.readAsArrayBuffer(file);
        }
        function sanitizeText(text) {
            return text.toString().replace(/^\uFEFF/, '').replace(/[\u0000-\u001F]/g, '').trim();
        }
        function showImportedWords() {
            const container = document.getElementById('importedWords');
            const display = wordPairs.slice(0, 30);
            container.innerHTML = display.map(p => `<div class="word-item">${p.word}<br>${p.meaning}</div>`).join('');
            if (wordPairs.length > 30) container.innerHTML += `<div class="word-item">…他 ${wordPairs.length-30} 組</div>`;
        }

        // ---------- 游戏核心 ----------
        function startGame() {
            let pool = wordPairs.length ? wordPairs : defaultWords;
            const count = Math.min(10, pool.length);
            const shuffledPool = shuffleArray([...pool]);
            const selected = shuffledPool.slice(0, count);
            if (selected.length === 0) {
                alert('単語がありません。先にインポートしてください。');
                return;
            }
            resetGame();          // 先重置，清空 remainingPairs
            remainingPairs = selected;  // 再赋值
            gameRunning = true;
            renderGameBoard();
            startTimer();
        }

        // 重新渲染左右列（根据 remainingPairs，左右分别打乱顺序）
        function renderGameBoard() {
            if (!gameRunning && remainingPairs.length > 0) return;
            // 准备左右卡片数据，分别打乱
            const wordCards = remainingPairs.map((p, idx) => ({
                pairId: idx,
                type: 'word',
                display: p.word,
                pair: p
            }));
            const meaningCards = remainingPairs.map((p, idx) => ({
                pairId: idx,
                type: 'meaning',
                display: p.meaning,
                pair: p
            }));
            shuffleArray(wordCards);
            shuffleArray(meaningCards);
            renderCol('leftCol', wordCards);
            renderCol('rightCol', meaningCards);
        }

        function renderCol(colId, cards) {
            const col = document.getElementById(colId);
            const title = col.querySelector('.col-title');
            col.innerHTML = '';
            col.appendChild(title);
            cards.forEach(cardData => {
                const card = document.createElement('div');
                card.className = 'card';
                card.dataset.pairId = cardData.pairId;
                card.dataset.type = cardData.type;
                card.textContent = cardData.display;
                card.addEventListener('click', () => handleCardClick(card));
                col.appendChild(card);
            });
        }

        function handleCardClick(card) {
            if (!gameRunning) return;
            if (card.classList.contains('selected') && selectedCard === card) {
                card.classList.remove('selected');
                selectedCard = null;
                return;
            }
            if (selectedCard) {
                const first = selectedCard;
                if (first === card) return;
                // 类型相同则切换选中
                if (first.dataset.type === card.dataset.type) {
                    first.classList.remove('selected');
                    card.classList.add('selected');
                    selectedCard = card;
                    return;
                }
                // 类型不同，配对
                if (first.dataset.pairId === card.dataset.pairId) {
                    // 配对成功
                    const pairId = parseInt(first.dataset.pairId);
                    // 从 remainingPairs 中移除该配对
                    remainingPairs = remainingPairs.filter((_, idx) => idx !== pairId);
                    correctCount++;
                    document.getElementById('correctCount').textContent = correctCount;
                    document.getElementById('correctSound').play();
                    // 移除选中状态
                    first.classList.remove('selected');
                    selectedCard = null;
                    // 重新渲染游戏板（剩余卡片重新排列对齐）
                    renderGameBoard();
                    // 检查是否完成
                    if (remainingPairs.length === 0) {
                        gameRunning = false;
                        clearInterval(timerInterval);
                        const timeStr = document.getElementById('timer').textContent;
                        updateLeaderboard();
                        setTimeout(() => {
                            alert(`🎉 ゲームクリア！\n時間: ${timeStr}\n正解数: ${correctCount}`);
                        }, 500);
                    }
                } else {
                    // 配对失败
                    first.classList.add('wrong');
                    card.classList.add('wrong');
                    first.classList.remove('selected');
                    setTimeout(() => {
                        first.classList.remove('wrong');
                        card.classList.remove('wrong');
                    }, 700);
                    selectedCard = null;
                }
            } else {
                card.classList.add('selected');
                selectedCard = card;
            }
        }

        // 工具函数
        function shuffleArray(array) {
            for (let i = array.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [array[i], array[j]] = [array[j], array[i]];
            }
            return array;
        }

        function startTimer() {
            startTime = Date.now();
            clearInterval(timerInterval);
            timerInterval = setInterval(() => {
                const seconds = Math.floor((Date.now() - startTime) / 1000);
                document.getElementById('timer').textContent = 
                    `${String(Math.floor(seconds/60)).padStart(2,'0')}:${String(seconds%60).padStart(2,'0')}`;
            }, 1000);
        }

        function resetGame() {
            correctCount = 0;
            selectedCard = null;
            document.getElementById('correctCount').textContent = '0';
            clearInterval(timerInterval);
            document.getElementById('timer').textContent = '00:00';
            document.getElementById('leftCol').innerHTML = '<div class="col-title">📖 単語</div>';
            document.getElementById('rightCol').innerHTML = '<div class="col-title">📝 意味</div>';
            gameRunning = false;
            remainingPairs = [];
        }

        // 排行榜
        function updateLeaderboard() {
            const playerName = document.getElementById('playerName').value.trim() || '匿名プレイヤー';
            const time = document.getElementById('timer').textContent;
            const records = JSON.parse(localStorage.getItem('wordGameRecords') || '[]');
            records.push({ name: playerName, score: correctCount, time: time, timestamp: Date.now() });
            records.sort((a, b) => b.score - a.score || a.timestamp - b.timestamp);
            localStorage.setItem('wordGameRecords', JSON.stringify(records.slice(0, 10)));
            showLeaderboard();
        }
        function showLeaderboard() {
            const records = JSON.parse(localStorage.getItem('wordGameRecords') || '[]');
            const tbody = document.querySelector('#leaderboardTable tbody');
            tbody.innerHTML = records.map((r, i) => `<tr><td>${i+1}</td><td>${r.name}</td><td>${r.score}</td><td>${r.time}</td></tr>`).join('');
        }

        // 页面加载默认显示
        window.onload = function() {
            if (wordPairs.length === 0) {
                wordPairs = [...defaultWords];
                showImportedWords();
            }
        };
    </script>
